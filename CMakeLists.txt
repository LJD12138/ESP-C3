# ==============================================================================
# ESP-IDF项目根CMakeLists.txt
# 项目名称：G02
# 说明：以下内容必须严格按照此顺序，否则cmake无法正常工作
# ==============================================================================

# 设置cmake最低版本要求
cmake_minimum_required(VERSION 3.16)

# ==============================================================================
# 自定义组件目录声明
# ==============================================================================
# Hardware  : 硬件相关组件（Key、Print等）
# Middlewares: 中间件组件（EasyFlash、EasyLogger、Protocol等）
# ComFunc   : 通用功能组件（I2C、校验、过滤等）
# ==============================================================================
set(EXTRA_COMPONENT_DIRS
    "Hardware"
    "Middlewares"
    "ComFunc"
)

# 引入ESP-IDF项目构建系统并声明项目
include($ENV{IDF_PATH}/tools/cmake/project.cmake)
project(G02)

# ==============================================================================
# 仅对 LVGL 组件启用 -O2:
# 全局保持 Debug(-Og) 调试体验, LVGL 软件渲染对优化等级敏感,
# -O2 下绘图吞吐约提升 1.5~2x, 直接影响显示帧率与动画流畅度
# ==============================================================================
if(TARGET __idf_lvgl__lvgl)
    target_compile_options(__idf_lvgl__lvgl PRIVATE -O2)
endif()
